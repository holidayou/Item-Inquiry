export default { async fetch(request, env) { const url = new URL(request.url); const action = url.searchParams.get('action') || 'query'; // CORS 预检 if (request.method === 'OPTIONS') { return new Response(null, { headers: { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': 'POST, GET, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type', }, }); } // 处理建议请求 if (action === 'suggest') { const query = url.searchParams.get('q'); if (!query) return jsonResponse({ suggestions: [] }); const apiKey = env.DEEPSEEK_API_KEY; if (!apiKey) return jsonResponse({ error: '未配置 API Key' }, 500); const prompt = `请根据用户输入“${query}”，返回最多5个可能的药品名称（仅名称，不要编号），用 JSON 数组格式，例如：["布洛芬","阿莫西林"]。如果输入是批准文号，返回包含该文号的产品名称。`; try { const suggestions = await callDeepSeek(prompt, apiKey); return jsonResponse({ suggestions: suggestions.slice(0, 5) }); } catch (err) { return jsonResponse({ suggestions: [] }); } } // 处理图片文字提取关键词 if (action === 'extract') { if (request.method !== 'POST') return jsonResponse({ error: 'Method not allowed' }, 405); let { text } = await request.json(); if (!text) return jsonResponse({ error: '缺少文字内容' }, 400); const apiKey = env.DEEPSEEK_API_KEY; if (!apiKey) return jsonResponse({ error: '未配置 API Key' }, 500); const prompt = `请从以下文字中提取药品名称或批准文号（如果有多个，只返回最可能的一个）。只返回提取到的名称或文号，不要其他内容。如果无法提取，返回空字符串。\n\n文字内容：${text}`; try { const keyword = await callDeepSeek(prompt, apiKey); return jsonResponse({ keyword: keyword.trim() }); } catch (err) { return jsonResponse({ keyword: '' }); } } // 原有的产品查询逻辑（POST 请求） if (request.method !== 'POST') { return jsonResponse({ error: 'Method not allowed' }, 405); } let productName; try { const body = await request.json(); productName = body.productName?.trim(); } catch (e) { return jsonResponse({ error: '无效的 JSON 请求体' }, 400); } if (!productName) { return jsonResponse({ error: '产品名称不能为空' }, 400); } const apiKey = env.DEEPSEEK_API_KEY; if (!apiKey) return jsonResponse({ error: '服务器未配置 API Key' }, 500); const prompt = `请帮我查询“${productName}”这款产品的详细信息，包括： 1. 生产厂家全称 2. 是否具备进出口资质（根据公开信息判断） 3. 功能属性：是保健功能还是治病功能（药品/医疗器械） 4. 补充说明（批准文号、使用注意事项等） 请用简洁的中文回答，每项一行，格式如下： 厂家：xxx 进出口资质：xxx 功能属性：xxx 补充：xxx`; try { const response = await fetch('https://api.deepseek.com/v1/chat/completions', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${apiKey}`, }, body: JSON.stringify({ model: 'deepseek-chat', messages: [ { role: 'system', content: '你是一个专业的产品信息查询助手，可以联网搜索最新公开数据。回答时请严格遵循用户要求的格式。' }, { role: 'user', content: prompt }, ], temperature: 0.3, enable_search: true, }), }); const data = await response.json(); if (!response.ok) throw new Error(data.error?.message || 'API 调用失败'); const aiReply = data.choices[0].message.content; const lines = aiReply.split('\n'); let result = { manufacturer: '未获取到', importExport: '未获取到', functionType: '未获取到', additional: '未获取到' }; for (const line of lines) { if (line.includes('厂家：')) result.manufacturer = line.split('厂家：')[1].trim(); else if (line.includes('进出口资质：')) result.importExport = line.split('进出口资质：')[1].trim(); else if (line.includes('功能属性：')) result.functionType = line.split('功能属性：')[1].trim(); else if (line.includes('补充：')) result.additional = line.split('补充：')[1].trim(); } return jsonResponse(result); } catch (error) { console.error(error); return jsonResponse({ error: error.message }, 500); } // 辅助函数 function jsonResponse(data, status = 200) { return new Response(JSON.stringify(data), { status, headers: { 'Content-Type': 'application/json', 'Access-Control-Allow-Origin': '*' }, }); } async function callDeepSeek(prompt, apiKey) { const resp = await fetch('https://api.deepseek.com/v1/chat/completions', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${apiKey}`, }, body: JSON.stringify({ model: 'deepseek-chat', messages: [{ role: 'user', content: prompt }], temperature: 0.3, }), }); const data = await resp.json(); if (!resp.ok) throw new Error(data.error?.message); return data.choices[0].message.content; } }, };
